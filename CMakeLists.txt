cmake_minimum_required(VERSION 3.24)
project(SlopStudio VERSION 0.13.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This standalone backend requires Windows x86-64. Debian audio/UI integration remains incomplete.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SLOPX_PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/windows-x64" CACHE PATH "Standalone output directory")
set(SLOPX_VST3_SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/vst3sdk" CACHE PATH "Official Steinberg VST3 SDK")
set(SLOPX_CLAP_SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/clap" CACHE PATH "Official CLAP headers")
if(NOT EXISTS "${SLOPX_VST3_SDK}/pluginterfaces/vst/ivstaudioprocessor.h" OR NOT EXISTS "${SLOPX_CLAP_SDK}/include/clap/clap.h")
    message(FATAL_ERROR "Official SDK headers are missing. Run bootstrap-dependencies.ps1 or set SLOPX_VST3_SDK and SLOPX_CLAP_SDK.")
endif()
find_program(SLOPX_CSC_EXECUTABLE NAMES csc.exe HINTS "$ENV{WINDIR}/Microsoft.NET/Framework64/v4.0.30319" REQUIRED)
file(MAKE_DIRECTORY "${SLOPX_PACKAGE_DIR}")
file(GLOB SLOPX_LUA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lua/src/*.c")
list(FILTER SLOPX_LUA_SOURCES EXCLUDE REGEX "/(lua|luac|onelua|ltests).c$")
add_library(SlopControl SHARED native/StudioControl.cpp ${SLOPX_LUA_SOURCES})
target_include_directories(SlopControl PRIVATE third_party/lua/src)
target_link_libraries(SlopControl PRIVATE dinput8 dxguid user32)
set_target_properties(SlopControl PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}" LIBRARY_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}")
if(NOT MSVC)
 target_link_options(SlopControl PRIVATE -static)
endif()
add_library(SlopAudio SHARED native/WasapiAudio.cpp)
target_link_libraries(SlopAudio PRIVATE ole32 uuid propsys)
if(MSVC)
 set_property(TARGET SlopAudio SlopControl PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set_target_properties(SlopAudio PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}" LIBRARY_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}")
if(NOT MSVC)
 target_link_options(SlopAudio PRIVATE -static)
endif()
add_library(SlopNative SHARED native/PluginHost.cpp)
target_include_directories(SlopNative PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ara" "${SLOPX_VST3_SDK}" "${SLOPX_CLAP_SDK}/include")
target_link_libraries(SlopNative PRIVATE ole32 user32)
set_target_properties(SlopNative PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}" LIBRARY_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}")
if(MSVC)
    target_compile_options(SlopNative PRIVATE /W3 /EHsc)
    set_property(TARGET SlopNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(SlopNative PRIVATE -Wno-pragma-pack)
    target_link_options(SlopNative PRIVATE -static)
endif()
option(SLOPX_GPL_COMPANIONS "Build separate GPL ASIO and Ableton Link companion" ON)
if(SLOPX_GPL_COMPANIONS)
 add_executable(SlopBridge companions/SlopBridge.cpp)
 target_include_directories(SlopBridge PRIVATE companions/third_party/asiosdk/ASIOSDK/common companions/third_party/link/include companions/third_party/link/modules/asio-standalone/include)
 target_compile_definitions(SlopBridge PRIVATE LINK_PLATFORM_WINDOWS ASIO_STANDALONE _WIN32_WINNT=0x0601)
 target_link_libraries(SlopBridge PRIVATE ole32 user32 advapi32 ws2_32 iphlpapi winmm)
 if(MSVC)
  set_property(TARGET SlopBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 endif()
 set_target_properties(SlopBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${SLOPX_PACKAGE_DIR}/Companions")
 if(NOT MSVC)
  target_link_options(SlopBridge PRIVATE -static -municode)
 endif()
endif()
file(GLOB SLOPX_APP_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cs")
set(SLOPX_APP "${SLOPX_PACKAGE_DIR}/SlopStudio.exe")
set(SLOPX_CSC_SOURCES)
foreach(FILE IN LISTS SLOPX_APP_SOURCES)
 file(TO_NATIVE_PATH "${FILE}" NATIVE_FILE)
 list(APPEND SLOPX_CSC_SOURCES "${NATIVE_FILE}")
endforeach()
file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest" SLOPX_MANIFEST)
file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/resources/SlopProX.ico" SLOPX_ICON)
set(MANAGED_REFERENCES /reference:System.Windows.Forms.dll /reference:System.Drawing.dll /reference:System.Web.Extensions.dll /reference:System.Xml.Linq.dll /reference:System.IO.Compression.dll /reference:System.IO.Compression.FileSystem.dll)
add_custom_command(OUTPUT "${SLOPX_APP}"
    COMMAND "${SLOPX_CSC_EXECUTABLE}" /nologo /target:winexe /platform:x64 /optimize+ "/out:${SLOPX_APP}" "/win32manifest:${SLOPX_MANIFEST}" "/win32icon:${SLOPX_ICON}" ${MANAGED_REFERENCES} ${SLOPX_CSC_SOURCES}
    DEPENDS ${SLOPX_APP_SOURCES} src/app.manifest resources/SlopProX.ico VERBATIM)
add_custom_target(SlopStudio ALL DEPENDS "${SLOPX_APP}")
add_dependencies(SlopStudio SlopNative SlopAudio SlopControl)
add_custom_target(RuntimeAssets ALL
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${SLOPX_PACKAGE_DIR}/Resources"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/resources/Startup.png" "${SLOPX_PACKAGE_DIR}/Resources/Startup.png"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/resources/STARTUP-ART-NOTICE.md" "${SLOPX_PACKAGE_DIR}/Resources/STARTUP-ART-NOTICE.md"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/src/SlopStudio.exe.config" "${SLOPX_PACKAGE_DIR}"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/locales" "${SLOPX_PACKAGE_DIR}/locales"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/docs" "${SLOPX_PACKAGE_DIR}/docs"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "${SLOPX_PACKAGE_DIR}/licenses"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/examples" "${SLOPX_PACKAGE_DIR}/Examples"
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/companions" "${SLOPX_PACKAGE_DIR}/Companions/Source"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/companions/LICENSE" "${SLOPX_PACKAGE_DIR}/Companions/LICENSE"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/THIRD-PARTY-NOTICES.md" "${CMAKE_CURRENT_SOURCE_DIR}/sdk-revisions.txt" "${SLOPX_PACKAGE_DIR}"
    VERBATIM)
file(GENERATE OUTPUT "${SLOPX_PACKAGE_DIR}/build-metadata.txt" CONTENT "Application=Slop Studio 0.13
Architecture=AMD64 PE32+
Target=Windows 11 x86-64 standalone
Configuration=$<CONFIG>
CMake=${CMAKE_VERSION}
Generator=${CMAKE_GENERATOR}
C++ compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}
C# compiler=${SLOPX_CSC_EXECUTABLE}
Runtime=.NET Framework 4.8; Windows UCRT/system DLLs
VST3=official SDK; see sdk-revisions.txt
CLAP=official headers; see sdk-revisions.txt
")
