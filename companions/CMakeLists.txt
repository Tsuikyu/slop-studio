cmake_minimum_required(VERSION 3.24)
project(SlopBridge VERSION 0.6.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "SlopBridge requires Windows x86-64.")
endif()
add_executable(SlopBridge SlopBridge.cpp)
target_compile_features(SlopBridge PRIVATE cxx_std_17)
target_include_directories(SlopBridge PRIVATE third_party/asiosdk/ASIOSDK/common third_party/link/include third_party/link/modules/asio-standalone/include)
target_compile_definitions(SlopBridge PRIVATE LINK_PLATFORM_WINDOWS ASIO_STANDALONE _WIN32_WINNT=0x0601)
target_link_libraries(SlopBridge PRIVATE ole32 user32 advapi32 ws2_32 iphlpapi winmm)
if(MINGW)
 target_link_options(SlopBridge PRIVATE -static -municode)
elseif(MSVC)
 set_property(TARGET SlopBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
